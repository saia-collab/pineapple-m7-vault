# PM7 GOOGLE DRIVE MEDIA REPURPOSING PIPELINE
$DriveFolder = "G:\My Drive\Pineapple-Media-HQ\01-INBOX"
$LocalVault = "C:\Pineapple Contractors M7\02_Media_Vault\Pineapple-Media-HQ\2026_05_GENERAL_INBOX"
$Outbox = "C:\Pineapple Contractors M7\01_Command_Center\Outbox_Drafts"
$DailyOversight = "H:\My Drive\Pineapple Contractors M7\_DAILY_OVERSIGHT\Outbox_Drafts"

# 1. Sync Raw Files from Google Drive
if (Test-Path $DriveFolder) {
    Get-ChildItem -Path $DriveFolder -Include *.mov,*.mp4,*.jpg -Recurse | ForEach-Object {
        $dest = Join-Path $LocalVault $_.Name
        if (-not (Test-Path $dest)) {
            Copy-Item -Path $_.FullName -Destination $dest -Force
            Write-Host "[SYNC] Pulled from Drive: $($_.Name)" -ForegroundColor Cyan
        }
    }
}

# 2. Run Brand Firewall Verification Pass
python "C:\Pineapple Contractors M7\04_Tech_Lab\scripts\brand_firewall.py" --check

# 3. Mirror Outbox Drafts to Google Drive Oversight for Approval
if (Test-Path $DailyOversight) {
    robocopy $Outbox $DailyOversight /MIR /NJH /NJS /NDL /nc /ns
    Write-Host "[SYNC] Outbox Drafts mirrored to Google Drive Oversight" -ForegroundColor Cyan
}

Write-Host "`n[PASS] Media Processed & Saved PAUSED in Outbox_Drafts!" -ForegroundColor Yellow
